if(POLICY CMP0135)
    cmake_policy(SET CMP0135 NEW)
endif()
set(CMAKE_POLICY_VERSION_MINIMUM 3.5)
include(FetchContent)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL ON CACHE BOOL "" FORCE)
FetchContent_Declare(lvgl URL https://github.com/lvgl/lvgl/archive/refs/tags/v9.5.0.tar.gz URL_HASH SHA256=34a955cdf3a2d005507b704e87357af669a114523b6d3f77b5344fdc68717bc6)
set(ENABLE_CUSTOM_COMPILER_FLAGS OFF CACHE BOOL "Avoid cJSON /Za conflict with MSVC C11" FORCE)
set(ENABLE_CJSON_TEST OFF CACHE BOOL "" FORCE)
set(ENABLE_CJSON_UTILS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(cjson URL https://github.com/DaveGamble/cJSON/archive/refs/tags/v1.7.19.tar.gz URL_HASH SHA256=7fa616e3046edfa7a28a32d5f9eacfd23f92900fe1f8ccd988c1662f30454562)
set(ZLIB_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
# The zlib.net fossil endpoint can return a proxy/error page under CI.
# Fetch the exact official zlib 1.3.1 commit instead of trusting a mutable URL.
FetchContent_Declare(zlib
    GIT_REPOSITORY https://github.com/madler/zlib.git
    GIT_TAG 51b7f2abdade71cd9bb0e7a373ef2610ec6f9daf
    GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(lvgl cjson zlib)
set(APP_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
file(GLOB FONT_SOURCES CONFIGURE_DEPENDS "${APP_ROOT}/main/fonts/*.c")
file(GLOB PNG_FILES CONFIGURE_DEPENDS "${APP_ROOT}/main/assets/png/*.png")
# Portable C arrays instead of GNU ld binary objects: works with MSVC as well.
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/app_asset_sizes.h" "#pragma once\n")
set(PNG_SOURCES)
foreach(PNG IN LISTS PNG_FILES)
    set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${PNG}")
    get_filename_component(STEM "${PNG}" NAME_WE)
    string(TOUPPER "${STEM}" UPPER)
    file(SIZE "${PNG}" BYTES)
    file(READ "${PNG}" HEX_DATA HEX)
    string(REGEX REPLACE "(..)" "0x\\1," C_DATA "${HEX_DATA}")
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${STEM}.c" "#include <stdint.h>\nconst uint8_t _binary_${STEM}_png_start[] = {${C_DATA}};\n")
    file(APPEND "${CMAKE_CURRENT_BINARY_DIR}/app_asset_sizes.h" "#define APP_PNG_${UPPER}_SIZE ${BYTES}U\n")
    list(APPEND PNG_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/${STEM}.c")
endforeach()
add_executable(zolee_simulator simulator.c
    "${APP_ROOT}/main/app_slave_display.c" "${APP_ROOT}/main/app_slave_display_input.c"
    "${APP_ROOT}/main/app_events.c" "${APP_ROOT}/main/app_diagnostics.c" "${APP_ROOT}/main/app_chart.c" "${APP_ROOT}/main/app_assets.c"
    "${APP_ROOT}/main/app_payload.c" "${APP_ROOT}/main/app_ingest.c"
    "${APP_ROOT}/main/app_state.c" "${APP_ROOT}/simulator/home_fixture.c" "${APP_ROOT}/simulator/battery_fixture.c" "${APP_ROOT}/main/app_battery_diagnostics.c" "${APP_ROOT}/main/app_energy_assistant.c" "${APP_ROOT}/main/app_templates.c" "${APP_ROOT}/main/app_template_zip.c" ${FONT_SOURCES} ${PNG_SOURCES})
target_include_directories(zolee_simulator BEFORE PRIVATE "${cjson_SOURCE_DIR}" "${CMAKE_CURRENT_SOURCE_DIR}/compat" "${APP_ROOT}/tests/stubs" "${APP_ROOT}/main" "${CMAKE_CURRENT_BINARY_DIR}")
target_include_directories(zolee_simulator PRIVATE "${zlib_SOURCE_DIR}" "${zlib_BINARY_DIR}")
target_compile_definitions(zolee_simulator PRIVATE APP_TEMPLATE_DESKTOP APP_TEMPLATE_DEFAULT_ROOT="${CMAKE_CURRENT_BINARY_DIR}/templates")
target_link_libraries(zolee_simulator PRIVATE lvgl cjson zlibstatic)
# Copy only on configure; runtime selection is stored beside these copies, never in source packages.
file(COPY "${APP_ROOT}/templates" DESTINATION "${CMAKE_CURRENT_BINARY_DIR}")
set_property(TARGET zolee_simulator PROPERTY VS_DEBUGGER_WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
if(MSVC)
    target_compile_options(zolee_simulator PRIVATE /utf-8 /W3)
    target_compile_definitions(zolee_simulator PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
if(WIN32)
    target_link_libraries(zolee_simulator PRIVATE user32 gdi32)
else()
    target_link_libraries(zolee_simulator PRIVATE m)
endif()
add_test(NAME simulator_demo COMMAND zolee_simulator --self-test)
